<script>
  let teachers = [];
  let students = [];
  let rawAppointments = []; 
  const periods = ["第一節", "第二節", "第三節", "第四節", "午休", "第五節", "第六節", "第七節", "第八節"];

  let selectedDateStr = "";
  let activeFilterType = 'teacher';
  let currentFilterValue = 'ALL';
  let currentSemester = 0;

  const basePalette = [
    { bg: '#fee2e2', border: '#fca5a5', text: '#991b1b' },
    { bg: '#dbeafe', border: '#93c5fd', text: '#1e40af' },
    { bg: '#dcfce7', border: '#86efac', text: '#166534' },
    { bg: '#f3e8ff', border: '#d8b4fe', text: '#6b21a8' },
    { bg: '#fef3c7', border: '#fde047', text: '#854d0e' },
    { bg: '#fce7f3', border: '#f472b6', text: '#9d174d' },
    { bg: '#cff4fc', border: '#a5f3fc', text: '#155e75' },
    { bg: '#ffedd5', border: '#fed7aa', text: '#9a3412' },
    { bg: '#e2e8f0', border: '#cbd5e1', text: '#334155' },
    { bg: '#ecfdf5', border: '#a7f3d0', text: '#065f46' }
  ];

  function openSidebar(side) {
    closeSidebars();
    if (side === 'left') {
      document.getElementById('sidebarLeft').classList.add('open');
    } else if (side === 'right') {
      document.getElementById('sidebarRight').classList.add('open');
    }
    document.getElementById('sidebarOverlay').classList.add('active');
  }

  function closeSidebars() {
    document.getElementById('sidebarLeft').classList.remove('open');
    document.getElementById('sidebarRight').classList.remove('open');
    document.getElementById('sidebarOverlay').classList.remove('active');
  }

  function showLoading(show) {
    document.getElementById('loadingOverlay').style.display = show ? 'flex' : 'none';
  }

  window.onload = function() {
    const currentMonth = new Date().getMonth();
    currentSemester = (currentMonth >= 1 && currentMonth <= 6) ? 2 : 1;
    fetchCloudData();
  };

  function fetchCloudData() {
    showLoading(true);
    google.script.run.withSuccessHandler(function(lists) {
      teachers = lists.teachers || [];
      students = lists.students || [];

      google.script.run.withSuccessHandler(function(data) {
        rawAppointments = data || [];
        showLoading(false);
        updateAllUI();
      }).getAppointments();

    }).getLists();
  }

  function getColorForEntity(name, type) {
    const list = (type === 'teacher') ? teachers : students;
    const index = list.indexOf(name);
    
    if (index === -1) return { bg: '#f1f5f9', border: '#cbd5e1', text: '#475569' };
    if (index < basePalette.length) return basePalette[index];

    const hue = (index * 137.5) % 360; 
    return {
      bg: `hsl(${hue}, 70%, 90%)`,
      border: `hsl(${hue}, 60%, 75%)`,
      text: `hsl(${hue}, 80%, 25%)`
    };
  }

  function loadSemesterCalendar(semester) {
    currentSemester = semester;
    renderCalendars();
    closeSidebars();
  }

  function setFilterType(type) {
    activeFilterType = type;
    currentFilterValue = 'ALL';
    document.getElementById('btnShowTeacher').className = `toggle-btn ${type === 'teacher' ? 'active' : ''}`;
    document.getElementById('btnShowStudent').className = `toggle-btn ${type === 'student' ? 'active' : ''}`;
    renderFilterButtons();
    renderCalendars();
  }

  function setFilterValue(val) {
    currentFilterValue = val;
    renderFilterButtons();
    renderCalendars();
  }

  function renderFilterButtons() {
    const container = document.getElementById('filterTagList');
    const syncContainer = document.getElementById('calendarSyncContainer');
    const list = activeFilterType === 'teacher' ? teachers : students;
    const label = activeFilterType === 'teacher' ? '全部教師' : '全部學生';

    let html = `<button class="filter-btn all" onclick="setFilterValue('ALL')" style="${currentFilterValue === 'ALL' ? 'outline:2px solid #0f766e;' : ''}">${label}</button>`;

    list.forEach(item => {
      const colors = getColorForEntity(item, activeFilterType);
      const isSelected = currentFilterValue === item;
      html += `<button class="filter-btn" onclick="setFilterValue('${item}')" style="background:${colors.bg}; border-color:${colors.border}; color:${colors.text}; ${isSelected ? 'box-shadow: 0 0 0 2px #0f766e;' : ''}">
                ${item}
              </button>`;
    });

    container.innerHTML = html;

    // 當選擇「單一教師」檢視時，提供同步到 Google 日曆的功能
    if (activeFilterType === 'teacher' && currentFilterValue !== 'ALL') {
      syncContainer.innerHTML = `
        <button class="btn-sync-gcal" onclick="syncTeacherToGoogleCalendar('${currentFilterValue}')">
          📅 同步「${currentFilterValue}」晤談至 Google 日曆
        </button>
      `;
    } else {
      syncContainer.innerHTML = '';
    }
  }

  function syncTeacherToGoogleCalendar(teacherName) {
  const teacherEvents = rawAppointments.filter(evt => evt.teacher === teacherName);
  
  if (teacherEvents.length === 0) {
    alert(`「${teacherName}」目前沒有任何晤談預約紀錄。`);
    return;
  }

  // 1. 讓教師填寫 Google 帳號
  const userEmail = prompt(`請輸入要同步行程的 Google 帳號 (Email)：`, "");
  
  if (userEmail === null) return; // 使用者取消
  
  const trimmedEmail = userEmail.trim();
  if (!trimmedEmail) {
    alert("請輸入有效的 Google 帳號！");
    return;
  }

  if (confirm(`確定要將「${teacherName}」的 ${teacherEvents.length} 筆晤談紀錄同步至 ${trimmedEmail} 的 Google 日曆嗎？`)) {
    showLoading(true);
    google.script.run
      .withSuccessHandler(function(result) {
        showLoading(false);
        if (result && result.success) {
          alert(`同步完成！已成功新增行程至 ${trimmedEmail} 的 Google 日曆。`);
        } else {
          alert(`同步失敗：${result ? result.message : '未知錯誤'}`);
        }
      })
      .withFailureHandler(function(err) {
        showLoading(false);
        alert(`同步發生錯誤：${err.message}`);
      })
      .syncTeacherEventsToCalendar(teacherName, teacherEvents, trimmedEmail);
  }
}

  function renderManagementTags() {
    const tList = document.getElementById('teacherTagList');
    const sList = document.getElementById('studentTagList');

    tList.innerHTML = teachers.map((t) => {
      const colors = getColorForEntity(t, 'teacher');
      return `<div class="manage-tag" style="background:${colors.bg}; border-color:${colors.border}; color:${colors.text};">
                ${t} <span class="delete-btn" onclick="removeTeacherByName('${t}')">&times;</span>

    }).join('');

    sList.innerHTML = students.map((s) => {
      const colors = getColorForEntity(s, 'student');
      return `<div class="manage-tag" style="background:${colors.bg}; border-color:${colors.border}; color:${colors.text};">
                ${s} <span class="delete-btn" onclick="removeStudentByName('${s}')">&times;</span>
              </div>`;
    }).join('');
  }

  function addTeacher() {
    const input = document.getElementById('newTeacherInput');
    const name = input.value.trim();
    if (name && !teachers.includes(name)) {
      teachers.push(name);
      input.value = '';
      syncListsToCloud();
    }
  }

  function removeTeacherByName(name) {
    teachers = teachers.filter(t => t !== name);
    syncListsToCloud();
  }

  function addStudent() {
    const input = document.getElementById('newStudentInput');
    const name = input.value.trim();
    if (name && !students.includes(name)) {
      students.push(name);
      input.value = '';
      syncListsToCloud();
    }
  }

  function removeStudentByName(name) {
    students = students.filter(s => s !== name);
    syncListsToCloud();
  }

  function syncListsToCloud() {
    showLoading(true);
    google.script.run.withSuccessHandler(function() {
      showLoading(false);
      updateAllUI();
    }).saveLists(teachers, students);
  }

  function confirmDeleteAppointment(id, event, infoStr) {
    if (event) event.stopPropagation();
    if (confirm(`確定要刪除這筆晤談紀錄嗎？\n\n【${infoStr}】`)) {
      showLoading(true);
      google.script.run.withSuccessHandler(function() {
        fetchCloudData();
      }).deleteAppointmentById(id);
    }
  }

  function updateAllUI() {
    renderManagementTags();
    initSelectOptions();
    renderFilterButtons();
    renderCalendars();
  }

  function initSelectOptions() {
    const tSelect = document.getElementById('teacherSelect');
    const pSelect = document.getElementById('periodSelect');
    const sSelect = document.getElementById('studentSelect');

    tSelect.innerHTML = '';
    pSelect.innerHTML = '';
    sSelect.innerHTML = '';

    teachers.forEach(t => tSelect.add(new Option(t, t)));
    students.forEach(s => sSelect.add(new Option(s, s)));
    periods.forEach(p => pSelect.add(new Option(p, p)));
  }

  function getMonthRange() {
    const now = new Date();
    let currentYear = now.getFullYear();
    let currentMonth = now.getMonth();

    if (currentSemester === 1) {
      let startYear = (currentMonth < 7) ? currentYear - 1 : currentYear;
      return [
        { year: startYear, month: 8 },
        { year: startYear, month: 9 },
        { year: startYear, month: 10 },
        { year: startYear, month: 11 },
        { year: startYear + 1, month: 0 }
      ];
    } else {
      let targetYear = (currentMonth >= 7) ? currentYear + 1 : currentYear;
      return [
        { year: targetYear, month: 1 },
        { year: targetYear, month: 2 },
        { year: targetYear, month: 3 },
        { year: targetYear, month: 4 },
        { year: targetYear, month: 5 }
      ];
    }
  }

  function renderCalendars() {
    const calendarList = document.getElementById('calendarList');
    calendarList.innerHTML = '';
    const months = getMonthRange();
    const colorMode = document.getElementById('colorModeSelect').value;

    const periodShortMap = {
      "第一節": "1", "第二節": "2", "第三節": "3", "第四節": "4",
      "午休": "午",
      "第五節": "5", "第六節": "6", "第七節": "7", "第八節": "8"
    };

    const isMobile = window.innerWidth <= 768;

    months.forEach(({ year, month }) => {
      const monthCard = document.createElement('div');
      monthCard.className = 'month-card';

      const monthTitle = document.createElement('div');
      monthTitle.className = 'month-title';
      monthTitle.innerText = `${year} 年 ${month + 1} 月`;
      monthCard.appendChild(monthTitle);

      const grid = document.createElement('div');
      grid.className = 'calendar-grid';

      const weekdays = ['日', '一', '二', '三', '四', '五', '六'];
      weekdays.forEach(day => {
        const header = document.createElement('div');
        header.className = 'weekday-header';
        header.innerText = day;
        grid.appendChild(header);
      });

      const firstDay = new Date(year, month, 1).getDay();
      const daysInMonth = new Date(year, month + 1, 0).getDate();
      const prevDaysInMonth = new Date(year, month, 0).getDate();

      for (let i = firstDay - 1; i >= 0; i--) {
        const emptyCell = document.createElement('div');
        emptyCell.className = 'day-cell other-month';
        emptyCell.innerHTML = `<div class="day-number">${prevDaysInMonth - i}</div>`;
        grid.appendChild(emptyCell);
      }

      for (let day = 1; day <= daysInMonth; day++) {
        const dayCell = document.createElement('div');
        dayCell.className = 'day-cell';
        
        const dateStr = `${year}-${String(month + 1).padStart(2, '0')}-${String(day).padStart(2, '0')}`;
        
        const dayNumber = document.createElement('div');
        dayNumber.className = 'day-number';
        dayNumber.innerText = day;
        dayCell.appendChild(dayNumber);

        const eventContainer = document.createElement('div');

        const dayEvents = rawAppointments.filter(evt => evt.date === dateStr);
        dayEvents.sort((a, b) => periods.indexOf(a.period) - periods.indexOf(b.period));

        dayEvents.forEach(evt => {
          if (currentFilterValue !== "ALL") {
            if (activeFilterType === 'teacher' && evt.teacher !== currentFilterValue) return;
            if (activeFilterType === 'student' && evt.student !== currentFilterValue) return;
          }

          const tag = document.createElement('div');
          tag.className = 'event-tag';
          
          const keyName = (colorMode === 'teacher') ? evt.teacher : evt.student;
          const colors = getColorForEntity(keyName, colorMode);
          
          tag.style.backgroundColor = colors.bg;
          tag.style.borderColor = colors.border;
          tag.style.color = colors.text;

          const infoStr = `${evt.date} ${evt.period} ${evt.teacher}-${evt.student}`;

          if (isMobile) {
            const shortPeriod = periodShortMap[evt.period] || evt.period;
            tag.innerHTML = `<span class="event-text"><b>${shortPeriod}</b> ${evt.teacher}-${evt.student}</span>`;
            
            tag.addEventListener('click', function(e) {
              e.preventDefault();
              e.stopPropagation();
              confirmDeleteAppointment(evt.id, e, infoStr);
            });

          } else {
            tag.innerHTML = `<span class="event-text">${evt.period} ${evt.teacher}-${evt.student}</span>
                             <span class="delete-btn" onclick="confirmDeleteAppointment('${evt.id}', event, '${infoStr}')">&times;</span>`;
          }
          
          eventContainer.appendChild(tag);
        });

        dayCell.appendChild(eventContainer);
        dayCell.onclick = () => openModal(dateStr);
        grid.appendChild(dayCell);
      }

      const totalCells = firstDay + daysInMonth;
      const nextDays = (7 - (totalCells % 7)) % 7;
      for (let day = 1; day <= nextDays; day++) {
        const emptyCell = document.createElement('div');
        emptyCell.className = 'day-cell other-month';
        emptyCell.innerHTML = `<div class="day-number">${day}</div>`;
        grid.appendChild(emptyCell);
      }

      monthCard.appendChild(grid);
      calendarList.appendChild(monthCard);
    });
  }

  function openModal(dateStr) {
    selectedDateStr = dateStr;
    document.getElementById('modalTitle').innerText = `${dateStr} 新增晤談紀錄`;
    document.getElementById('eventModal').style.display = 'flex';
  }

  function closeModal() {
    document.getElementById('eventModal').style.display = 'none';
  }

  function saveEvent() {
    const teacher = document.getElementById('teacherSelect').value;
    const period = document.getElementById('periodSelect').value;
    const student = document.getElementById('studentSelect').value;

    if (!teacher || !student) {
      alert("請選擇教師與學生！");
      return;
    }

    const eventData = {
      date: selectedDateStr,
      teacher: teacher,
      period: period,
      student: student
    };

    closeModal();
    showLoading(true);

    google.script.run.withSuccessHandler(function() {
      fetchCloudData();
    }).addAppointment(eventData);
  }
